\section{\makebox[\widthof{\faUsers}][c]{\color{CVBlue}\faUsers}\ 项目经历}

% --- 第一个项目（核心，置顶） ---
\textbf{基于边缘AI的心电信号实时预警监护系统} \hfill 2026.03 -- 2026.07 \titlebreak
项目描述：针对传统床旁监护误报率高、云端依赖断网失效等临床痛点，本项目基于\textbf{边缘AI}与\textbf{TinyML}技术，在\textbf{STM32H7}嵌入式平台部署轻量化心电预警模型，融合多源生理信号实现急性心脏事件的实时精准识别。
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{模型压缩与部署}：采用\textbf{知识蒸馏、剪枝与量化}技术对AI模型进行轻量化优化，在STM32H7平台实现ECG信号的低延迟、低功耗实时推理。
    \item \textbf{多模态融合预警}：融合\textbf{ECG与PPG}多源生理信号，设计\textbf{决策级融合算法}，精准识别室颤、室性早搏等危急事件，有效降低单一传感器失效导致的误报率。
    \item \textbf{算法验证}：在\textbf{MIT-BIH、MIMIC}等权威数据库上完成模型训练与验证，计划交付具备床旁报警接口的嵌入式原型系统，为危重症快速响应救治提供技术保障。
\end{itemize}

% --- 第二个项目 ---
\textbf{MiniSQL数据库引擎设计} \hfill 2025.03 -- 2025.06 \titlebreak
项目描述：基于\textbf{C++}设计并实现一个支持标准SQL子集的单用户关系型数据库引擎。
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{存储与索引}：自主设计\textbf{内存池与数据页管理}机制，减少内存碎片；实现基于\textbf{B+树}的聚簇索引系统，解决节点分裂与合并（Coalesce/Redistribute）难题。
    \item \textbf{查询执行}：开发查询解析器与执行器，支持SELECT/INSERT/DELETE等标准操作，通过\textbf{缓冲区管理策略}降低磁盘I/O频率。
    \item \textbf{性能表现}：在10万级数据量测试中，插入与查询平均耗时控制在\textbf{100ms以内}，性能优于同课程80\%的小组方案。
\end{itemize}

% --- 第三个项目 ---
\textbf{智能图像资源管理系统} \hfill 2025.10 -- 2026.01 \titlebreak
项目描述：基于JavaScript全栈开发集成\textbf{深度学习模型}的图片管理系统。
\begin{itemize}[nosep, topsep=\intraProjectListTopSep]
    \item \textbf{后端与AI集成}：设计SQL数据库存储图片元数据，集成深度学习模型自动提取图片特征，生成\textbf{5个高精度语义标签}，构建自动化素材分类管线。
    \item \textbf{前端与并发优化}：开发RESTful API与前端交互界面，支持基于内容的模糊检索；利用\textbf{异步任务队列}处理大图分析与标签生成，优化系统并发处理能力。
    \item \textbf{项目成果}：实现了素材标签识别的自动化，显著提升了资源检索效率。
\end{itemize}
